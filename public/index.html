} /** Does the actual multi-calendar fetch, filter, and AM-resolution work. */ function fetchClientEventsFromCalendars_() { const now = new Date(); const start = new Date(now.getTime() - DAYS_PAST * 86400000); if (!parsed) return null; const am = resolveAM_(parsed.client); const key = event.getTitle() + '|' + event.getStartTime().toISOString(); return { id: event.getId(), key: key, // stable identifier for claiming — survives across re-fetches, unlike getId() client: parsed.client, meetingType: parsed.meetingType, am: am, amEmail: AM_EMAIL_MAP[am] || '', title: parsed.title, start: event.getStartTime().toISOString(), end: event.getEndTime().toISOString(), allDay: event.isAllDayEvent(), location: event.getLocation() || '', description: event.getDescription() || '', }; }) .filter(Boolean); // drops events with no [Meeting Type] tag — i.e. non-client meetings return { clientEvents: clientEvents, errors: errors }; } /** * Handles claim/release actions from the "In Process" drag-and-drop board. * Browser sends Content-Type: text/plain to avoid a CORS preflight (Apps * Script doesn't support OPTIONS), so the JSON body is parsed manually here. * * Expected body: { action: 'claim'|'release', key: '', * client: '...', meetingType: '...', title: '...', * start: '...', name: '' } */ function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonResponse_({ error: 'Invalid request body' }); } if (!body.key || !body.action) { return jsonResponse_({ error: 'Missing key or action' }); } const sheet = getOrCreateClaimsSheet_(); const rowNum = findClaimRow_(sheet, body.key); if (body.action === 'claim') { if (!body.name) return jsonResponse_({ error: 'Missing claimant name' }); const now = new Date().toISOString();
